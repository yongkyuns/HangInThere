cmake_minimum_required(VERSION 3.22.1)
project(hanginthere_hdr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
find_program(PYTHON_EXECUTABLE python3 REQUIRED)
set(HDR_BUILD "${REPO_ROOT}/build/android-hdr")
execute_process(COMMAND "${PYTHON_EXECUTABLE}" "${REPO_ROOT}/scripts/build-android-hdr.py"
    --ndk "${ANDROID_NDK}" --abi "${ANDROID_ABI}" --output "${HDR_BUILD}"
    RESULT_VARIABLE HDR_BUILD_RESULT)
if(NOT HDR_BUILD_RESULT EQUAL 0)
    message(FATAL_ERROR "Pinned HDR decoder dependencies failed to build")
endif()
foreach(LIBRARY avcodec avformat avfilter avutil swscale)
    add_library(${LIBRARY} SHARED IMPORTED)
    set_target_properties(${LIBRARY} PROPERTIES IMPORTED_LOCATION "${HDR_BUILD}/${ANDROID_ABI}/jni/lib${LIBRARY}.so")
endforeach()
add_library(hanginthere_hdr SHARED HdrDecoder.cpp HdrDecoderJni.cpp)
target_include_directories(hanginthere_hdr PRIVATE "${HDR_BUILD}/${ANDROID_ABI}/install/include")
target_compile_options(hanginthere_hdr PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(hanginthere_hdr PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(hanginthere_hdr PRIVATE avformat avcodec avfilter avutil swscale)
target_link_options(hanginthere_hdr PRIVATE "-Wl,-z,max-page-size=16384")

# Pixel matching shares the exact Apple sampler/geometry; only bitmap ingress differs.
find_package(OpenCV REQUIRED CONFIG)
add_library(hanginthere_camera SHARED BackgroundCameraJni.cpp)
target_include_directories(hanginthere_camera PRIVATE "${REPO_ROOT}/native/apple/Sources/CObservedRailPixels")
target_link_libraries(hanginthere_camera PRIVATE OpenCV::opencv_java4)
target_compile_options(hanginthere_camera PRIVATE -Wall -Wextra -Werror)
target_link_options(hanginthere_camera PRIVATE "-Wl,-z,max-page-size=16384")
